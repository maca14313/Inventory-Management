   200→// Custom Alphabetical Month-Year Date Picker
   201→const MONTHS = ['January','February','March','April','May','June','July','August','September','October','November','December']
   202→function CustomDatePicker({ value, onChange, compact }: { value: string; onChange: (v: string) => void; compact?: boolean }) {
   203→  const [open, setOpen] = useState(false)
   204→  const today = new Date()
   205→  const [year, setYear] = useState(today.getFullYear())
   206→  const [month, setMonth] = useState(today.getMonth())
   207→  const parsed = value ? new Date(value + 'T00:00:00') : null
   208→
   209→  const years = Array.from({ length: 10 }, (_, i) => today.getFullYear() - 2 + i)
   210→  // Months sorted alphabetically
   211→  const alphaMonths = MONTHS.map((m, i) => ({ name: m, index: i })).sort((a, b) => a.name.localeCompare(b.name))
   212→
   213→  const daysInMonth = new Date(year, month + 1, 0).getDate()
   214→  const days = Array.from({ length: daysInMonth }, (_, i) => i + 1)
   215→
   216→  const selectDate = (d: number) => {
   217→    const iso = `${year}-${String(month + 1).padStart(2, '0')}-${String(d).padStart(2, '0')}`
   218→    onChange(iso)
   219→    setOpen(false)
   220→  }
   221→
   222→  return (
   223→    <div className="relative">
   224→      <button type="button" onClick={() => setOpen(!open)}
   225→        className={`w-full flex items-center gap-2 rounded-lg border border-input bg-background text-foreground hover:bg-accent transition ${compact ? 'px-2 py-1 text-xs' : 'px-3 py-2 text-sm'}`}>
   226→        <Calendar className={`${compact ? 'w-3.5 h-3.5' : 'w-4 h-4'} text-muted-foreground shrink-0`} />
   227→        {parsed ? parsed.toLocaleDateString() : <span className="text-muted-foreground">Pick a date</span>}
   228→      </button>
   229→      {open && (
   230→        <div className="absolute z-50 mt-1 w-64 rounded-xl border bg-popover p-3 shadow-xl">
   231→          <div className="flex gap-2 mb-3">
   232→            <Select value={String(year)} onValueChange={v => setYear(parseInt(v))}>
   233→              <SelectTrigger className="h-8 text-xs flex-1"><SelectValue /></SelectTrigger>
   234→              <SelectContent>{years.map(y => <SelectItem key={y} value={String(y)}>{y}</SelectItem>)}</SelectContent>
   235→            </Select>
   236→            <Select value={String(month)} onValueChange={v => setMonth(parseInt(v))}>
   237→              <SelectTrigger className="h-8 text-xs flex-1"><SelectValue /></SelectTrigger>
   238→              <SelectContent className="max-h-48">
   239→                {alphaMonths.map(m => <SelectItem key={m.index} value={String(m.index)}>{m.name}</SelectItem>)}
   240→              </SelectContent>
   241→            </Select>
   242→          </div>
   243→          <div className="grid grid-cols-7 gap-1">
   244→            {days.map(d => {
   245→              const iso = `${year}-${String(month + 1).padStart(2, '0')}-${String(d).padStart(2, '0')}`
   246→              const isSelected = value === iso
   247→              return <button key={d} onClick={() => selectDate(d)}
   248→                className={`h-7 w-7 rounded-md text-xs font-medium transition ${isSelected ? 'bg-blue-600 text-white' : 'hover:bg-accent text-foreground'}`}>{d}</button>
   249→            })}
   250→          </div>
   251→        </div>
   252→      )}
   253→    </div>
   254→  )
   255→}

   283→// ─── File preview thumbnail
   284→// Renders an actual image thumbnail for image files using URL.createObjectURL().
   285→// For non-image files, falls back to a FileText icon on a muted tile.
   286→// The object URL is computed synchronously via useMemo (no cascading renders)
   287→// and revoked in a useEffect cleanup when the file changes or the component unmounts
   288→// to prevent memory leaks.
   289→function FilePreviewThumb({ file, size = 32 }: { file: File; size?: number }) {
   290→  const isImage = file.type.startsWith('image/')
   291→  const url = useMemo(() => {
   292→    if (!isImage) return null
   293→    return URL.createObjectURL(file)
   294→  }, [file, isImage])
   295→  useEffect(() => {
   296→    if (!url) return
   297→    return () => URL.revokeObjectURL(url)
   298→  }, [url])
   299→  if (!isImage || !url) {
   300→    return (
   301→      <div
   302→        className="flex items-center justify-center bg-muted/60 rounded shrink-0 border"
   303→        style={{ width: size, height: size }}
   304→      >

   475→// ─── PaginatedCombobox: reusable Popover-based combobox with search + pagination.
   476→// Used to replace every large-data <Select> (warehouses, categories, products, suppliers,
   477→// enterprises, branches) so that the dropdown stays fast and usable when the list grows.
   478→// Style mirrors the Returns warehouse combobox (violet accent for selected).
   479→type ComboItem = { value: string; label: string; sublabel?: string; badge?: string }
   480→function PaginatedCombobox({
   481→  items,
   482→  value,
   483→  onChange,
   484→  placeholder = 'Select...',
   485→  allLabel,
   486→  allValue = '_all',
   487→  perPage = 10,
   488→  triggerClassName = '',
   489→  contentClassName = 'w-44',
   490→  disabled = false,
   491→  compact = false,
   492→}: {
   493→  items: ComboItem[]
   494→  value: string | undefined | ''
   495→  onChange: (value: string) => void
   496→  placeholder?: string
   497→  allLabel?: string
   498→  allValue?: string
   499→  perPage?: number
   500→  triggerClassName?: string
   501→  contentClassName?: string
   502→  disabled?: boolean
   503→  compact?: boolean
   504→}) {
   505→  const [open, setOpen] = useState(false)
   506→  const [search, setSearch] = useState('')
   507→  const [page, setPage] = useState(1)
   508→
   509→  // Resolve the currently selected item's label for the trigger button.
   510→  const selectedLabel = (() => {
   511→    if (allLabel && (value === allValue || value === '' || value === undefined)) return allLabel
   512→    const it = items.find(i => i.value === value)
   513→    if (it) return it.label
   514→    return ''
   515→  })()
   516→
   517→  const filtered = useMemo(() => {
   518→    const s = search.trim().toLowerCase()
   519→    if (!s) return items
   520→    return items.filter(i =>
   521→      i.label.toLowerCase().includes(s)
   522→      || (i.sublabel || '').toLowerCase().includes(s)
   523→      || (i.badge || '').toLowerCase().includes(s)
   524→    )
   525→  }, [items, search])
   526→
   527→  const totalPages = Math.max(1, Math.ceil(filtered.length / perPage))
   528→  const safePage = Math.min(page, totalPages)
   529→  const paged = filtered.slice((safePage - 1) * perPage, safePage * perPage)
   530→
   531→  const handlePick = (v: string) => {
   532→    onChange(v)
   533→    setOpen(false)
   534→    setSearch('')
   535→    setPage(1)
   536→  }
   537→
   538→  const onOpenChange = (o: boolean) => {
   539→    setOpen(o)
   540→    if (!o) { setSearch(''); setPage(1) }
   541→  }
   542→
   543→  const sizeCls = compact ? 'h-7 text-[11px] px-2' : 'h-8 text-xs px-2'
   544→
   545→  return (
   546→    <Popover open={open} onOpenChange={onOpenChange}>
   547→      <PopoverTrigger asChild>
   548→        <Button
   549→          type="button"
   550→          variant="outline"
   551→          role="combobox"
   552→          disabled={disabled}
   553→          className={`${sizeCls} justify-between w-full font-normal ${triggerClassName}`}
   554→        >
   555→          <span className="truncate">{selectedLabel || placeholder}</span>
   556→          <ChevronDown className="w-3 h-3 opacity-50 shrink-0" />
   557→        </Button>
   558→      </PopoverTrigger>
   559→      <PopoverContent className={`${contentClassName} p-0`} align="start">
   560→        {/* Sticky search */}
   561→        <div className="p-1.5 border-b sticky top-0 bg-popover z-10">
   562→          <div className="relative">
   563→            <Search className="w-3 h-3 text-muted-foreground absolute left-1.5 top-1/2 -translate-y-1/2" />
   564→            <Input
   565→              placeholder="Search..."
   566→              value={search}
   567→              onChange={e => { setSearch(e.target.value); setPage(1) }}
   568→              className="h-7 text-xs pl-6"
   569→            />
   570→          </div>
   571→        </div>
   572→        {/* Scrollable list */}
   573→        <div className="max-h-56 overflow-y-auto">
   574→          {allLabel && (
   575→            <button
   576→              type="button"
   577→              onClick={() => handlePick(allValue)}
   578→              className={`w-full text-left px-2 py-1.5 text-xs hover:bg-violet-50 dark:hover:bg-violet-500/10 ${(value === allValue || value === '' || value === undefined) ? 'bg-violet-50 dark:bg-violet-500/10 font-medium text-violet-700 dark:text-violet-400' : ''}`}
   579→            >
   580→              {allLabel}
   581→            </button>
   582→          )}
   583→          {paged.map(it => (
   584→            <button
   585→              key={it.value}
   586→              type="button"
   587→              onClick={() => handlePick(it.value)}
   588→              className={`w-full text-left px-2 py-1.5 text-xs hover:bg-violet-50 dark:hover:bg-violet-500/10 flex items-center justify-between gap-1.5 ${value === it.value ? 'bg-violet-50 dark:bg-violet-500/10 font-medium text-violet-700 dark:text-violet-400' : ''}`}
   589→            >
   590→              <span className="truncate">{it.label}</span>
   591→              {it.sublabel && <span className="text-[9px] text-muted-foreground font-mono shrink-0">{it.sublabel}</span>}
   592→              {it.badge && <span className="text-[9px] text-muted-foreground shrink-0">{it.badge}</span>}
   593→            </button>
   594→          ))}
   595→          {filtered.length === 0 && (
   596→            <div className="px-2 py-2 text-[11px] text-muted-foreground text-center">No matches.</div>
   597→          )}
   598→        </div>
   599→        {/* Pagination footer (only when more than 1 page) */}
   600→        {totalPages > 1 && (
   601→          <div className="flex items-center justify-between gap-1 px-1.5 py-1 border-t bg-popover">
   602→            <span className="text-[10px] text-muted-foreground">{safePage}/{totalPages} · {filtered.length} items</span>
   603→            <div className="flex items-center gap-0.5">
   604→              <Button size="sm" variant="outline" disabled={safePage <= 1} onClick={() => setPage(p => Math.max(1, p - 1))} className="h-5 w-5 p-0"><ChevronLeft className="w-2.5 h-2.5" /></Button>
   605→              <Button size="sm" variant="outline" disabled={safePage >= totalPages} onClick={() => setPage(p => Math.min(totalPages, p + 1))} className="h-5 w-5 p-0"><ChevronRight className="w-2.5 h-2.5" /></Button>
   606→            </div>
   607→          </div>
   608→        )}
   609→      </PopoverContent>
   610→    </Popover>
   611→  )
   612→}
