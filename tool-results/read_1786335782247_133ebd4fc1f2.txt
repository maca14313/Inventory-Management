    63→// Custom Alphabetical Month-Year Date Picker
    64→const MONTHS = ['January','February','March','April','May','June','July','August','September','October','November','December']
    65→function CustomDatePicker({ value, onChange, compact }: { value: string; onChange: (v: string) => void; compact?: boolean }) {
    66→  const [open, setOpen] = useState(false)
    67→  const today = new Date()
    68→  const [year, setYear] = useState(today.getFullYear())
    69→  const [month, setMonth] = useState(today.getMonth())
    70→  const parsed = value ? new Date(value + 'T00:00:00') : null
    71→
    72→  const years = Array.from({ length: 10 }, (_, i) => today.getFullYear() - 2 + i)
    73→  // Months sorted alphabetically
    74→  const alphaMonths = MONTHS.map((m, i) => ({ name: m, index: i })).sort((a, b) => a.name.localeCompare(b.name))
    75→
    76→  const daysInMonth = new Date(year, month + 1, 0).getDate()
    77→  const days = Array.from({ length: daysInMonth }, (_, i) => i + 1)
    78→
    79→  const selectDate = (d: number) => {
    80→    const iso = `${year}-${String(month + 1).padStart(2, '0')}-${String(d).padStart(2, '0')}`
    81→    onChange(iso)
    82→    setOpen(false)
    83→  }
    84→
    85→  return (
    86→    <div className="relative">
    87→      <button type="button" onClick={() => setOpen(!open)}
    88→        className={`w-full flex items-center gap-2 rounded-lg border border-input bg-background text-foreground hover:bg-accent transition ${compact ? 'px-2 py-1 text-xs' : 'px-3 py-2 text-sm'}`}>
    89→        <Calendar className={`${compact ? 'w-3.5 h-3.5' : 'w-4 h-4'} text-muted-foreground shrink-0`} />
    90→        {parsed ? parsed.toLocaleDateString() : <span className="text-muted-foreground">Pick a date</span>}
    91→      </button>
    92→      {open && (
    93→        <div className="absolute z-50 mt-1 w-64 rounded-xl border bg-popover p-3 shadow-xl">
    94→          <div className="flex gap-2 mb-3">
    95→            <Select value={String(year)} onValueChange={v => setYear(parseInt(v))}>
    96→              <SelectTrigger className="h-8 text-xs flex-1"><SelectValue /></SelectTrigger>
    97→              <SelectContent>{years.map(y => <SelectItem key={y} value={String(y)}>{y}</SelectItem>)}</SelectContent>
    98→            </Select>
    99→            <Select value={String(month)} onValueChange={v => setMonth(parseInt(v))}>
   100→              <SelectTrigger className="h-8 text-xs flex-1"><SelectValue /></SelectTrigger>
   101→              <SelectContent className="max-h-48">
   102→                {alphaMonths.map(m => <SelectItem key={m.index} value={String(m.index)}>{m.name}</SelectItem>)}
   103→              </SelectContent>
   104→            </Select>
   105→          </div>
   106→          <div className="grid grid-cols-7 gap-1">
   107→            {days.map(d => {
   108→              const iso = `${year}-${String(month + 1).padStart(2, '0')}-${String(d).padStart(2, '0')}`
   109→              const isSelected = value === iso
   110→              return <button key={d} onClick={() => selectDate(d)}
   111→                className={`h-7 w-7 rounded-md text-xs font-medium transition ${isSelected ? 'bg-blue-600 text-white' : 'hover:bg-accent text-foreground'}`}>{d}</button>
   112→            })}
   113→          </div>
   114→        </div>
   115→      )}
   116→    </div>
   117→  )
   118→}
