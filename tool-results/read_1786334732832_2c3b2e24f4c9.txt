     1→/**
     2→ * Multi-Tenant Enterprise IMS — Database Driver
     3→ *
     4→ * HIERARCHY ENFORCEMENT:
     5→ *   Super Admin      → sees everything, manages all enterprises
     6→ *   Enterprise Admin  → scoped to their enterpriseId, manages branches/users/products/orders within it
     7→ *   Branch Admin      → scoped to their branchId, manages warehouses/users/orders within it
     8→ *   Warehouse Admin   → scoped to their warehouseId, manages stock/orders within it
     9→ *
    10→ * All findMany/findById/create/update/remove operations accept an AuthContext
    11→ * that enforces tenant boundaries. A warehouse_admin can NEVER see data from
    12→ * another warehouse, even if they guess the ID.
    13→ */

   166→// ─── Tenant Scoping Helpers ─────────────────────────────────
   167→
   168→function scopeEnterpriseId(ctx: AuthContext): number | null {
   169→  if (ctx.role === 'super_admin') return null // sees all
   170→  return ctx.enterpriseId
   171→}
   172→
   173→function scopeBranchId(ctx: AuthContext): number | null {
   174→  if (ctx.role === 'super_admin' || ctx.role === 'enterprise_admin') return null
   175→  return ctx.branchId
   176→}
   177→
   178→function scopeWarehouseId(ctx: AuthContext): number | null {
   179→  if (ctx.role === 'warehouse_admin') return ctx.warehouseId
   180→  return null
   181→}
   182→
   183→/** Check if a user context can access a specific enterprise */
   184→function canAccessEnterprise(ctx: AuthContext, eid: number): boolean {
   185→  if (ctx.role === 'super_admin') return true
   186→  return ctx.enterpriseId === eid
   187→}
   188→
   189→/** Check if a user context can access a specific branch */
   190→function canAccessBranch(ctx: AuthContext, eid: number, bid: number): boolean {
   191→  if (!canAccessEnterprise(ctx, eid)) return false
   192→  if (ctx.role === 'super_admin' || ctx.role === 'enterprise_admin') return true
   193→  return ctx.branchId === bid
   194→}
   195→
   196→/** Check if a user context can access a specific warehouse */
   197→function canAccessWarehouse(ctx: AuthContext, eid: number, bid: number, wid: number): boolean {
   198→  if (!canAccessBranch(ctx, eid, bid)) return false
   199→  if (ctx.role === 'warehouse_admin') return ctx.warehouseId === wid
   200→  return true
   201→}
   202→
   203→/** Check if actor can manage a target user based on hierarchy */
   204→function canManageUser(ctx: AuthContext, target: User): boolean {
   205→  if (ctx.role === 'super_admin') return true
   206→  if (ctx.role === 'enterprise_admin') {
   207→    // Can only manage users in same enterprise, and only lower roles
   208→    if (target.enterpriseId !== ctx.enterpriseId) return false
   209→    const roleOrder: Record<Role, number> = { super_admin: 0, enterprise_admin: 1, branch_admin: 2, warehouse_admin: 3 }
   210→    return roleOrder[ctx.role] < roleOrder[target.role]
   211→  }
   212→  if (ctx.role === 'branch_admin') {
   213→    // Can only manage users in same branch, only warehouse_admins
   214→    if (target.branchId !== ctx.branchId) return false
   215→    return target.role === 'warehouse_admin'
   216→  }
   217→  return false // warehouse_admin cannot manage any users
   218→}

   220→// ─── Persistence ─────────────────────────────────────────
   221→
   222→const isServer = typeof window === 'undefined'
   223→let DATA_FILE = ''
   224→
   225→function persist() {
   226→  if (!isServer) return // Only persist on server side
   227→  try {
   228→    const fs = require('fs')
   229→    const path = require('path')
   230→    if (!DATA_FILE) DATA_FILE = path.join(process.cwd(), 'db', 'ims-data.json')
   231→    const data = {
   232→      nextId,
   233→      enterprises, features, branches, warehouses, users,
   234→      categories, suppliers, products, stock, orders, orderItems,
   235→      auditLogs, batches, stockOutRecorders,
   236→    }
   237→    fs.writeFileSync(DATA_FILE, JSON.stringify(data, null, 2), 'utf-8')
   238→  } catch (e) { console.error('Persist error:', e) }
   239→}
   240→
   241→function loadPersisted() {
   242→  if (!isServer) return false // Only load on server side
   243→  try {
   244→    const fs = require('fs')
   245→    const path = require('path')
   246→    DATA_FILE = path.join(process.cwd(), 'db', 'ims-data.json')
   247→    if (!fs.existsSync(DATA_FILE)) return false
   248→    const raw = fs.readFileSync(DATA_FILE, 'utf-8')
   249→    if (!raw || raw.trim().length < 10) return false
   250→    const data = JSON.parse(raw)
   251→    if (!data.enterprises || !Array.isArray(data.enterprises)) return false
   252→    nextId = data.nextId || 200
   253→    // Restore all arrays (splice + push to keep same references)
   254→    const restore = <T>(arr: T[], src: T[]) => { arr.splice(0, arr.length, ...src) }
   255→    restore(enterprises, data.enterprises)
   256→    restore(features, data.features || [])
   257→    restore(branches, data.branches || [])
   258→    restore(warehouses, data.warehouses || [])
   259→    restore(users, data.users || [])
   260→    restore(categories, data.categories || [])
   261→    restore(suppliers, data.suppliers || [])
   262→    restore(products, data.products || [])
   263→    restore(stock, data.stock || [])
   264→    restore(orders, data.orders || [])
   265→    restore(orderItems, data.orderItems || [])
   266→    restore(auditLogs, data.auditLogs || [])
   267→    restore(batches, data.batches || [])
   268→    restore(stockOutRecorders, data.stockOutRecorders || [])
   269→    console.log(`[DB] Loaded persisted data: ${enterprises.length} enterprises, ${products.length} products, ${stock.length} stock levels`)
   270→    return true
   271→  } catch (e) { console.error('Load persisted error:', e); return false }
   272→}

'+' : ''}${d.delta} ${prod?.unit || 'pcs'} of ${prod?.name || `Product #${d.productId}`} in ${wh.name}. New total: ${result.quantity}`, createdAt: now() })

  1100→      const i = batches.findIndex(b => b.id === id)
  1101→      if (i === -1) throw new Error('Not found')
  1102→      if (ctx.role !== 'super_admin' && batches[i].enterpriseId !== ctx.enterpriseId) throw new Error('FORBIDDEN')
  1103→      batches.splice(i, 1)
  1104→      persist()
  1105→    },
  1106→  },
  1107→
  1108→  // ── Stock Out Recorders ────────────────────────────────────
  1109→  stockOut: {
  1110→    findMany: async (ctx: AuthContext, eid?: number, wid?: number): Promise<StockOutRecorder[]> => {
  1111→      const targetEid = eid || (ctx.role !== 'super_admin' ? ctx.enterpriseId : undefined)
  1112→      let result = stockOutRecorders
  1113→      if (ctx.role === 'super_admin') {
  1114→        if (targetEid) result = result.filter(r => r.enterpriseId === targetEid)
  1115→        if (wid) result = result.filter(r => r.warehouseId === wid)
  1116→        return [...result]
  1117→      }
  1118→      if (!ctx.enterpriseId) return []
  1119→      result = result.filter(r => r.enterpriseId === ctx.enterpriseId)
  1120→      if (ctx.role === 'branch_admin') {
  1121→        const whIds = warehouses.filter(w => w.branchId === ctx.branchId).map(w => w.id)
  1122→        result = result.filter(r => whIds.includes(r.warehouseId))
  1123→      }
  1124→      if (ctx.role === 'warehouse_admin') {
  1125→        result = result.filter(r => r.warehouseId === ctx.warehouseId)
  1126→      }
  1127→      if (wid) result = result.filter(r => r.warehouseId === wid)
  1128→      return result
  1129→    },
  1130→    findById: async (ctx: AuthContext, id: number): Promise<StockOutRecorder | null> => {
  1131→      const r = stockOutRecorders.find(r => r.id === id)
  1132→      if (!r) return null
  1133→      if (ctx.role !== 'super_admin' && r.enterpriseId !== ctx.enterpriseId) return null
  1134→      return r
  1135→    },
  1136→    create: async (ctx: AuthContext, d: Partial<StockOutRecorder>): Promise<StockOutRecorder> => {
  1137→      if (ctx.role !== 'super_admin' && d.enterpriseId !== ctx.enterpriseId) throw new Error('FORBIDDEN')
  1138→      const r: StockOutRecorder = {
  1139→        id: ++nextId, enterpriseId: d.enterpriseId!, warehouseId: d.warehouseId!,
  1140→        recorderCode: d.recorderCode || `OUT-${Date.now()}`,
  1141→        personName: d.personName || '', personPhone: d.personPhone || '',
  1142→        personEmail: d.personEmail || '', personAddress: d.personAddress || '',
  1143→        notes: d.notes || '',
  1144→        status: 'draft', items: [],
  1145→        createdBy: ctx.userId, createdAt: now(), updatedAt: now(),
  1146→      }
  1147→      stockOutRecorders.push(r)
  1148→      persist()
  1149→      return r
  1150→    },
  1151→    update: async (ctx: AuthContext, id: number, d: Partial<StockOutRecorder>): Promise<StockOutRecorder> => {
  1152→      const i = stockOutRecorders.findIndex(r => r.id === id)
  1153→      if (i === -1) throw new Error('Not found')
  1154→      if (ctx.role !== 'super_admin' && stockOutRecorders[i].enterpriseId !== ctx.enterpriseId) throw new Error('FORBIDDEN')
  1155→      Object.assign(stockOutRecorders[i], d, { updatedAt: now() })
  1156→      persist()
  1157→      return stockOutRecorders[i]
  1158→    },
  1159→    addItem: async (ctx: AuthContext, recorderId: number, d: Partial<StockOutItem>): Promise<StockOutItem> => {
  1160→      const r = stockOutRecorders.find(r => r.id === recorderId)
  1161→      if (!r) throw new Error(`Stock-out recorder #${recorderId} not found. It may have been deleted.`)
  1162→      if (ctx.role !== 'super_admin' && r.enterpriseId !== ctx.enterpriseId) throw new Error('FORBIDDEN')
  1163→      const qty = d.quantity || 0
  1164→      if (qty <= 0) throw new Error('Quantity must be positive')
  1165→      if (!d.productId) throw new Error('Product is required')
  1166→      if (!d.warehouseId) throw new Error('Warehouse is required')
  1167→      // Check and deduct stock
  1168→      const si = stock.findIndex(s => s.productId === d.productId! && s.warehouseId === d.warehouseId!)
  1169→      if (si === -1) {
  1170→        const prod = products.find(p => p.id === d.productId)
  1171→        const wh = warehouses.find(w => w.id === d.warehouseId)
  1172→        throw new Error(`No stock found for ${prod?.name || `product #${d.productId}`} in ${wh?.name || `warehouse #${d.warehouseId}`}. Add stock first before stock-out.`)
  1173→      }
  1174→      if (stock[si].quantity < qty) throw new Error(`Insufficient stock: only ${stock[si].quantity} available for ${products.find(p=>p.id===d.productId)?.name||`product #${d.productId}`}, requested ${qty}`)
  1175→      stock[si].quantity -= qty
  1176→      stock[si].updatedAt = now()
  1177→      const item: StockOutItem = {
  1178→        id: ++nextId, recorderId, productId: d.productId!, warehouseId: d.warehouseId!,
  1179→        quantity: qty, unit: d.unit || 'pcs',
  1180→        isReturnable: d.isReturnable ?? false, returnDeadline: d.returnDeadline || null,
  1181→        notes: d.notes || '', createdAt: now(),
  1182→      }
  1183→      r.items.push(item)
  1184→      r.updatedAt = now()
  1185→      // Audit log
  1186→      const prod = products.find(p => p.id === d.productId!)
  1187→      auditLogs.push({ id: ++nextId, userId: ctx.userId, enterpriseId: r.enterpriseId, action: 'stock_out.add_item', entity: 'StockOutItem', entityId: item.id, details: `Stock out ${qty} ${d.unit || 'pcs'} of ${prod?.name || `Product #${d.productId}`} from warehouse #${d.warehouseId} (recorder #${recorderId}). Stock was ${stock[si].quantity + qty}, now ${stock[si].quantity}`, createdAt: now() })
  1188→      persist()
  1189→      return item
  1190→    },
  1191→    /** Remove a stock-out item and RESTORE the deducted stock */
  1192→    removeItem: async (ctx: AuthContext, recorderId: number, itemId: number): Promise<void> => {
  1193→      const r = stockOutRecorders.find(r => r.id === recorderId)
  1194→      if (!r) throw new Error(`Stock-out recorder #${recorderId} not found`)
  1195→      if (ctx.role !== 'super_admin' && r.enterpriseId !== ctx.enterpriseId) throw new Error('FORBIDDEN')
  1196→      const itemIdx = r.items.findIndex(i => i.id === itemId)
  1197→      if (itemIdx === -1) throw new Error(`Item #${itemId} not found in recorder #${recorderId}`)
  1198→      const item = r.items[itemIdx]
  1199→      const prod = products.find(p => p.id === item.productId)
  1200→      // Restore the stock that was deducted
  1201→      const si = stock.findIndex(s => s.productId === item.productId && s.warehouseId === item.warehouseId)
  1202→      if (si !== -1) {
  1203→        stock[si].quantity += item.quantity
  1204→        stock[si].updatedAt = now()
  1205→      } else {
  1206→        // If stock entry was deleted somehow, create it back
  1207→        stock.push({ id: ++nextId, productId: item.productId, warehouseId: item.warehouseId, quantity: item.quantity, createdAt: now(), updatedAt: now() })
  1208→      }
  1209→      r.items.splice(itemIdx, 1)
  1210→      r.updatedAt = now()
  1211→      // Audit log
  1212→      auditLogs.push({ id: ++nextId, userId: ctx.userId, enterpriseId: r.enterpriseId, action: 'stock_out.remove_item', entity: 'StockOutItem', entityId: itemId, details: `Removed stock-out item: restored ${item.quantity} ${item.unit} of ${prod?.name || `Product #${item.productId}`} to warehouse #${item.warehouseId}. Stock restored.`, createdAt: now() })
  1213→      persist()
  1214→    },
  1215→    remove: async (ctx: AuthContext, id: number): Promise<void> => {
  1216→      const i = stockOutRecorders.findIndex(r => r.id === id)
  1217→      if (i === -1) throw new Error(`Stock-out recorder #${id} not found`)
  1218→      if (ctx.role !== 'super_admin' && stockOutRecorders[i].enterpriseId !== ctx.enterpriseId) throw new Error('FORBIDDEN')
  1219→      const r = stockOutRecorders[i]
  1220→      // Restore stock for all items in the recorder being deleted
  1221→      // (Unlike cancellation, deletion implies the recorder was made in error)
  1222→      for (const item of r.items) {
  1223→        const si = stock.findIndex(s => s.productId === item.productId && s.warehouseId === item.warehouseId)
  1224→        if (si !== -1) {
  1225→          stock[si].quantity += item.quantity
  1226→          stock[si].updatedAt = now()
  1227→        } else {
  1228→          stock.push({ id: ++nextId, productId: item.productId, warehouseId: item.warehouseId, quantity: item.quantity, createdAt: now(), updatedAt: now() })
  1229→        }
  1230→      }
  1231→      if (r.items.length > 0) {
  1232→        auditLogs.push({ id: ++nextId, userId: ctx.userId, enterpriseId: r.enterpriseId, action: 'stock_out.delete_recorder', entity: 'StockOutRecorder', entityId: id, details: `Deleted recorder ${r.recorderCode} and restored ${r.items.length} item(s) to stock`, createdAt: now() })
  1233→      }
  1234→      stockOutRecorders.splice(i, 1)
  1235→      persist()
  1236→    },
  1237→  },
