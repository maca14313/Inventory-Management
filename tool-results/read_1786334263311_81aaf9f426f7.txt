     1→=== api.ts ===
     2→/**
     3→ * API Helper — typed fetch wrapper with auth context headers
     4→ */
     5→
     6→import { useAuthStore } from './auth-store'
     7→
     8→const BASE = process.env.NEXT_PUBLIC_BASE_PATH || ''
     9→
    10→export async function apiFetch<T>(
    11→  path: string,
    12→  options: RequestInit = {}
    13→): Promise<T> {
    14→  const headers = useAuthStore.getState().getAuthHeaders()
    15→  const res = await fetch(`${BASE}/api${path}`, {
    16→    ...options,
    17→    headers: {
    18→      'Content-Type': 'application/json',
    19→      ...headers,
    20→      ...(options.headers || {}),
    21→    },
    22→  })
    23→  if (!res.ok) {
    24→    const err = await res.json().catch(() => ({ error: res.statusText }))
    25→    throw new Error(err.error || `API Error ${res.status}`)
    26→  }
    27→  return res.json()
    28→}
    29→
    30→/** Parse AuthContext from request headers (server-side) */
    31→export function getAuthContextFromHeaders(headers: Headers) {
    32→  return {
    33→    userId: parseInt(headers.get('x-user-id') || '0', 10),
    34→    role: (headers.get('x-user-role') || 'warehouse_admin') as 'super_admin' | 'enterprise_admin' | 'branch_admin' | 'warehouse_admin',
    35→    enterpriseId: headers.get('x-enterprise-id') && headers.get('x-enterprise-id') !== 'null' ? parseInt(headers.get('x-enterprise-id')!, 10) : null,
    36→    branchId: headers.get('x-branch-id') && headers.get('x-branch-id') !== 'null' ? parseInt(headers.get('x-branch-id')!, 10) : null,
    37→    warehouseId: headers.get('x-warehouse-id') && headers.get('x-warehouse-id') !== 'null' ? parseInt(headers.get('x-warehouse-id')!, 10) : null,
    38→  }
    39→}

    62→export interface AuthActions {
    63→  login: (user: {
    64→    id: number; username: string; fullName: string; email: string
    65→    role: RoleLevel; enterpriseId: number | null; branchId: number | null
    66→    warehouseId: number | null; isActive: boolean
    67→  }) => void
    68→  logout: () => void
    69→  getAuthHeaders: () => Record<string, string>
    70→}
    71→
    72→const initialState: AuthState = {
    73→  userId: null, username: '', fullName: '', email: '',
    74→  role: null, enterpriseId: null, branchId: null, warehouseId: null,
    75→  isActive: false, isLoggedIn: false,
    76→}
    77→
    78→export const useAuthStore = create<AuthState & AuthActions>()((set, get) => ({
    79→  ...initialState,
    80→
    81→  login: (user) => set({
    82→    userId: user.id, username: user.username, fullName: user.fullName,
    83→    email: user.email, role: user.role, enterpriseId: user.enterpriseId,
    84→    branchId: user.branchId, warehouseId: user.warehouseId,
    85→    isActive: user.isActive, isLoggedIn: true,
    86→  }),
    87→
    88→  logout: () => set(initialState),
    89→
    90→  getAuthHeaders: () => {
    91→    const s = get()
    92→    return {
    93→      'x-user-id': String(s.userId || ''),
    94→      'x-user-role': s.role || '',
    95→      'x-enterprise-id': String(s.enterpriseId || ''),
    96→      'x-branch-id': String(s.branchId || ''),
    97→      'x-warehouse-id': String(s.warehouseId || ''),
    98→    }
    99→  },
   100→}))
   101→=== rbac.ts ===
   102→/**
   103→ * Role-Based Access Control (RBAC) — Strict Hierarchy Enforcement
   104→ *
   105→ * HIERARCHY (top → bottom):
   106→ *   super_admin      → Manages ALL enterprises. Grants access. Creates enterprise_admins.
   107→ *   enterprise_admin → Manages ONLY their enterprise. Creates branch_admins & warehouse_admins within it.
   108→ *   branch_admin     → Manages ONLY their branch. Creates warehouse_admins within it.
   109→ *   warehouse_admin  → Manages ONLY their warehouse. Stock adjustments, order views.
   110→ *
   111→ * KEY RULES:
   112→ *   1. Each role can ONLY manage roles BELOW them in the hierarchy
   113→ *   2. Each role is SCOPED to their tenant (enterprise/branch/warehouse)
   114→ *   3. Data access is filtered by tenant — you NEVER see cross-tenant data
   115→ *   4. Enterprise Admin cannot see other enterprises, even read-only
   116→ *   5. Branch Admin cannot see other branches in the same enterprise
   117→ *   6. Warehouse Admin cannot see other warehouses in the same branch
   118→ */
   119→

   265→// ─── Check Functions ─────────────────────────────────────────
   266→
   267→export function hasPermission(role: RoleLevel, perm: Permission): boolean {
   268→  return ROLE_PERMISSIONS[role]?.includes(perm) ?? false
   269→}
   270→
   271→export function getPermissions(role: RoleLevel): Permission[] {
   272→  return ROLE_PERMISSIONS[role] || []
   273→}
   274→
   275→export function getRoleLevel(role: RoleLevel): number {
   276→  return ROLE_HIERARCHY.indexOf(role)
   277→}
   278→
   279→/** Can actorRole manage (create/edit/delete) targetRole? */
   280→export function canManageRole(actorRole: RoleLevel, targetRole: RoleLevel): boolean {
   281→  return MANAGEABLE_ROLES[actorRole]?.includes(targetRole) ?? false
   282→}
   283→
   284→/** Is actorRole strictly above targetRole in hierarchy? */
   285→export function isAbove(actorRole: RoleLevel, targetRole: RoleLevel): boolean {
   286→  return getRoleLevel(actorRole) < getRoleLevel(targetRole)
   287→}
