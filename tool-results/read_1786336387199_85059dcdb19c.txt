    74→const MONTHS = ['January','February','March','April','May','June','July','August','September','October','November','December']
    75→function CustomDatePicker({ value, onChange, compact }: { value: string; onChange: (v: string) => void; compact?: boolean }) {
    76→  const [open, setOpen] = useState(false)
    77→  const today = new Date()
    78→  const [year, setYear] = useState(today.getFullYear())
    79→  const [month, setMonth] = useState(today.getMonth())
    80→  const parsed = value ? new Date(value + 'T00:00:00') : null
    81→
    82→  const years = Array.from({ length: 10 }, (_, i) => today.getFullYear() - 2 + i)
    83→  // Months sorted alphabetically
    84→  const alphaMonths = MONTHS.map((m, i) => ({ name: m, index: i })).sort((a, b) => a.name.localeCompare(b.name))
    85→
    86→  const daysInMonth = new Date(year, month + 1, 0).getDate()
    87→  const days = Array.from({ length: daysInMonth }, (_, i) => i + 1)
    88→
    89→  const selectDate = (d: number) => {
    90→    const iso = `${year}-${String(month + 1).padStart(2, '0')}-${String(d).padStart(2, '0')}`
    91→    onChange(iso)
    92→    setOpen(false)
    93→  }
    94→
    95→  return (
    96→    <div className="relative">
    97→      <button type="button" onClick={() => setOpen(!open)}
    98→        className={`w-full flex items-center gap-2 rounded-lg border border-input bg-background text-foreground hover:bg-accent transition ${compact ? 'px-2 py-1 text-xs' : 'px-3 py-2 text-sm'}`}>
    99→        <Calendar className={`${compact ? 'w-3.5 h-3.5' : 'w-4 h-4'} text-muted-foreground shrink-0`} />
   100→        {parsed ? parsed.toLocaleDateString() : <span className="text-muted-foreground">Pick a date</span>}
   101→      </button>
   102→      {open && (
   103→        <div className="absolute z-50 mt-1 w-64 rounded-xl border bg-popover p-3 shadow-xl">
   104→          <div className="flex gap-2 mb-3">
   105→            <Select value={String(year)} onValueChange={v => setYear(parseInt(v))}>
   106→              <SelectTrigger className="h-8 text-xs flex-1"><SelectValue /></SelectTrigger>
   107→              <SelectContent>{years.map(y => <SelectItem key={y} value={String(y)}>{y}</SelectItem>)}</SelectContent>
   108→            </Select>
   109→            <Select value={String(month)} onValueChange={v => setMonth(parseInt(v))}>
   110→              <SelectTrigger className="h-8 text-xs flex-1"><SelectValue /></SelectTrigger>
   111→              <SelectContent className="max-h-48">
   112→                {alphaMonths.map(m => <SelectItem key={m.index} value={String(m.index)}>{m.name}</SelectItem>)}
   113→              </SelectContent>
   114→            </Select>
   115→          </div>
   116→          <div className="grid grid-cols-7 gap-1">
   117→            {days.map(d => {
   118→              const iso = `${year}-${String(month + 1).padStart(2, '0')}-${String(d).padStart(2, '0')}`
   119→              const isSelected = value === iso
   120→              return <button key={d} onClick={() => selectDate(d)}
   121→                className={`h-7 w-7 rounded-md text-xs font-medium transition ${isSelected ? 'bg-blue-600 text-white' : 'hover:bg-accent text-foreground'}`}>{d}</button>
   122→            })}
   123→          </div>
   124→        </div>
   125→      )}
   126→    </div>
   127→  )
   128→}
