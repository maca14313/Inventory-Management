     5→ *   Super Admin       → sees everything, manages all enterprises

    20→ *     - db.stock.adjust()         → Lot source='adjustment' (for increase delta)

   249→function canManageUser(ctx: AuthContext, target: { role: string; enterpriseId: number | null; branchId: number | null }): boolean {
   250→  const targetRole = target.role as Role
   251→  if (ctx.role === 'super_admin') return true
   252→  const roleOrder: Record<Role, number> = { super_admin: 0, enterprise_admin: 1, branch_admin: 2, warehouse_admin: 3 }
   253→  if (ctx.role === 'enterprise_admin') {
   254→    if (target.enterpriseId !== ctx.enterpriseId) return false
   255→    return roleOrder[ctx.role] < roleOrder[targetRole]
   256→  }
   257→  if (ctx.role === 'branch_admin') {
   258→    if (target.branchId !== ctx.branchId) return false
   259→    return targetRole === 'warehouse_admin'
   260→  }
   261→  return false
   262→}

   341→/** Return list of warehouse IDs accessible to ctx */
   342→async function getAccessibleWarehouseIds(ctx: AuthContext, tx?: Tx): Promise<number[]> {
   343→  const client = tx ?? prisma
   344→  if (ctx.role === 'super_admin') {
   345→    const ws = await client.warehouse.findMany({ select: { id: true } })
   346→    return ws.map(w => w.id)
   347→  }
   348→  if (ctx.role === 'enterprise_admin') {
   349→    if (!ctx.enterpriseId) return []
   350→    const ws = await client.warehouse.findMany({ where: { enterpriseId: ctx.enterpriseId }, select: { id: true } })
   351→    return ws.map(w => w.id)
   352→  }
   353→  if (ctx.role === 'branch_admin') {
   354→    if (!ctx.branchId) return []
   355→    const ws = await client.warehouse.findMany({ where: { branchId: ctx.branchId }, select: { id: true } })
   356→    return ws.map(w => w.id)
   357→  }
   358→  return ctx.warehouseId ? [ctx.warehouseId] : []
   359→}

   363→/** Convert Prisma model rows to plain serializable objects (Date → ISO string).
   364→ *  Returns `unknown` so callers MUST cast to their target type — TypeScript
   365→ *  cannot verify the Date→string conversion structurally. */
   366→function ser(row: unknown): unknown {
   367→  if (row === null || row === undefined) return row
   368→  if (Array.isArray(row)) return row.map(ser)
   369→  if (row instanceof Date) return row.toISOString()
   370→  if (typeof row === 'object') {
   371→    const out: Record<string, unknown> = {}
   372→    for (const k of Object.keys(row as Record<string, unknown>)) {
   373→      const v = (row as Record<string, unknown>)[k]
   374→      out[k] = v instanceof Date ? v.toISOString() : (v && typeof v === 'object' ? ser(v) : v)
   375→    }
   376→    return out
   377→  }
   378→  return row
   379→}

   383→async function generateLotNumber(tx: Tx): Promise<string> {
   384→  const year = new Date().getFullYear()
   385→  const prefix = `LOT-${year}-`
   386→  const count = await tx.lot.count({ where: { lotNumber: { startsWith: prefix } } })
   387→  return `${prefix}${String(count + 1).padStart(4, '0')}`

   439→/**
   440→ * Deduct `quantity` from open lots of (productId, warehouseId) using FIFO.
   441→ * Sets closedAt when a lot's remainingQty hits 0.
   442→ * Throws if total open remainingQty < quantity.
   443→ */
   444→async function deductLotsFIFO(
   445→  tx: Tx,
   446→  ctx: AuthContext,
   447→  productId: number,
   448→  warehouseId: number,
   449→  quantity: number,
   450→): Promise<void> {
   451→  if (quantity <= 0) return
   452→  const openLots = await tx.lot.findMany({
   453→    where: {
   454→      productId,
   455→      warehouseId,
   456→      remainingQty: { gt: 0 },
   457→      closedAt: null,
   458→    },
   459→    orderBy: { receivedDate: 'asc' },
   460→  })
   461→  const totalOpen = openLots.reduce((s, l) => s + l.remainingQty, 0)
   462→  if (totalOpen < quantity) {
   463→    throw new Error(
   464→      `Insufficient open lot quantity for FIFO deduction (product #${productId}, warehouse #${warehouseId}): needed ${quantity}, available ${totalOpen}`,
   465→    )
   466→  }
   467→  let remaining = quantity
   468→  for (const lot of openLots) {
   469→    if (remaining <= 0) break
   470→    const take = Math.min(lot.remainingQty, remaining)
   471→    const newRemaining = lot.remainingQty - take
   472→    await tx.lot.update({
   473→      where: { id: lot.id },
   474→      data: {
   475→        remainingQty: newRemaining,
   476→        closedAt: newRemaining === 0 ? new Date() : null,
   477→      },
   478→    })
   479→    remaining -= take
   480→  }
   481→  await tx.auditLog.create({
   482→    data: {
   483→      userId: ctx.userId,
   484→      enterpriseId: ctx.enterpriseId,
   485→      action: 'lot.deduct_fifo',
   486→      entity: 'Lot',
   487→      entityId: null,
   488→      details: `FIFO-deducted ${quantity} from product #${productId} / warehouse #${warehouseId}`,
   489→    },
   490→  })
   491→}
   492→
   493→/**
   494→ * Reverse a lot by sourceRef:
   495→ *   - If remainingQty === quantity (untouched), delete the lot.
   496→ *   - Else, reduce both quantity and remainingQty by `quantity`.
   497→ *
   498→ * If the lot was partially consumed (remainingQty < quantity), we can only
   499→ * restore what's still on the lot (the rest has been shipped out). Caller
   500→ * must adjust stock independently.
   501→ */
   502→async function reverseLot(
   503→  tx: Tx,
   504→  ctx: AuthContext,
   505→  sourceRef: string,
   506→  quantity: number,
   507→): Promise<void> {
   508→  const lot = await tx.lot.findFirst({ where: { sourceRef } })
   509→  if (!lot) {
   510→    // No lot to reverse — silent no-op (defensive).
   511→    return
   512→  }
   513→  if (lot.remainingQty === lot.quantity && lot.quantity === quantity) {
   514→    // Untouched: delete entirely.
   515→    await tx.lot.delete({ where: { id: lot.id } })
   516→    await tx.auditLog.create({
   517→      data: {
   518→        userId: ctx.userId,
   519→        enterpriseId: lot.enterpriseId,
   520→        action: 'lot.reverse_delete',
   521→        entity: 'Lot',
   522→        entityId: lot.id,
   523→        details: `Deleted untouched lot ${lot.lotNumber} (sourceRef=${sourceRef})`,
   524→      },
   525→    })
   526→    return
   527→  }
   528→  // Partially consumed: reduce both quantity & remainingQty by `quantity`
   529→  // (but never below 0).
   530→  const newQty = Math.max(0, lot.quantity - quantity)
   531→  const newRemaining = Math.max(0, lot.remainingQty - quantity)
   532→  await tx.lot.update({
   533→    where: { id: lot.id },
   534→    data: {
   535→      quantity: newQty,
   536→      remainingQty: newRemaining,
   537→      closedAt: newRemaining === 0 ? (lot.closedAt ?? new Date()) : null,
   538→    },
   539→  })
   540→  await tx.auditLog.create({
   541→    data: {
   542→      userId: ctx.userId,
   543→      enterpriseId: lot.enterpriseId,
   544→      action: 'lot.reverse_reduce',
   545→      entity: 'Lot',
   546→      entityId: lot.id,
   547→      details: `Reduced lot ${lot.lotNumber} by ${quantity} (sourceRef=${sourceRef}); new qty=${newQty}, remaining=${newRemaining}`,
   548→    },
   549→  })
   550→}

   554→async function generateRecorderCode(tx: Tx, prefix: string, field: 'recorderCode'): Promise<string> {
   555→  const year = new Date().getFullYear()
   556→  const fullPrefix = `${prefix}-${year}-`
   557→  // Count existing recorders with this prefix
   558→  const count =
   559→    prefix === 'SIN'
   560→      ? await tx.stockInRecorder.count({ where: { recorderCode: { startsWith: fullPrefix } } })
   561→      : await tx.stockOutRecorder.count({ where: { recorderCode: { startsWith: fullPrefix } } })
   562→  return `${fullPrefix}${String(count + 1).padStart(4, '0')}`
   563→}

  1398→            action: 'stock_adjust', entity: 'Stock', entityId: s.id,
